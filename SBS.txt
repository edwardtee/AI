feature_range = range(2, max_features)

# Lists to store performance metrics
hist_num_features = []
hist_accuracy = []
hist_features_list = []

print("Starting Feature Optimization Loop via SBS...")

for k in feature_range:
    print(f"--> Testing subset of {k} features...")
    
    # Initialize SBS for the current k
    sbs = SequentialFeatureSelector(
        estimator=heart_net,    # Optimizing directly for your stacking model
        n_features_to_select=k,
        direction='backward',
        cv=3,                   # Keep low to save time during selection
        n_jobs=-1               # Use all CPU cores
    )
    
    # Fit SBS and transform the data
    sbs.fit(X_train, y_train)
    X_train_k = sbs.transform(X_train)
    X_test_k = sbs.transform(X_test)
    
    # Train the final Stacking Classifier on this specific subset
    heart_net.fit(X_train_k, y_train)
    
    # Evaluate
    y_pred_k = heart_net.predict(X_test_k)
    acc_k = accuracy_score(y_test, y_pred_k)
    
    # Save results
    hist_num_features.append(k)
    hist_accuracy.append(acc_k)
    hist_features_list.append(list(XX.columns[sbs.get_support()]))

# --- ADDED: Manually evaluate the baseline with ALL features ---
